    coordinates {(\ccnum{lineup.#1.usd},\ccnum{lineup.#1.dev_per_pair})};
  \node[font=\scriptsize\ttfamily, color=#2, #4]
    at (axis cs:\ccnum{lineup.#1.usd},\ccnum{lineup.#1.dev_per_pair}) {\ccnum{lineup.#1.name}};
}
